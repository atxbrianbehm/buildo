try {
 const {createBuildingConstructionGraph}=await server.ssrLoadModule("/src/features/building-family/contracts/bayConstructionGraph.ts");

 const graphs={legacy:createBuildingConstructionGraph(),c7:withConstructionWindowLayout(createBuildingConstructionGraph(),"window",0),off:withConstructionStorefront(createBuildingConstructionGraph(),0),storefront:withConstructionStorefront(createBuildingConstructionGraph(),1)};

  cases[name]={graph,resultHash:await hashCanonicalJson(result),geometryHash:await hashCanonicalJson(geometry),metrics:result.ir.metrics,instanceGroups:result.ir.instanceBatches.map(b=>({id:b.batchId,slot:b.materialSlotId,count:b.count})),semanticPaths:result.ir.semanticIndex.map(e=>e.semanticPath)};
 }
 if(cases.off.geometryHash!==cases.c7.geometryHash) throw new Error("Off geometry differs from C7.");

 writeFileSync(new URL("./compiled-cases.json",import.meta.url),JSON.stringify({schemaVersion:"0.1.0",cases,offMatchesC7:true,legacyExact:true,rendered:false,productIntegrated:false},null,2)+"\n");
 console.log("STOREFRONT_P1_PROBE",JSON.stringify({caseCount:4,offMatchesC7:true,legacyExact:true,storefrontMetrics:cases.storefront.metrics,frameGroups:cases.storefront.instanceGroups.filter(b=>b.slot==="frame.primary").map(b=>b.count)}));
} finally {await server.close();}
