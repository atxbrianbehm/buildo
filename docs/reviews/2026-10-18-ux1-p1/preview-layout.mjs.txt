import {createServer} from 'vite';import {chromium} from '@playwright/test';import {mkdir,writeFile} from 'node:fs/promises';
const out='test-results/ux1-p1-2026-10-18/layout-01';await mkdir(out,{recursive:false});const server=await createServer({cacheDir:out+'/.vite',server:{host:'127.0.0.1',port:0}});let browser;
try{await server.listen();browser=await chromium.launch({channel:'chromium',args:['--enable-unsafe-webgpu']});const page=await browser.newPage({viewport:{width:1493,height:1054}});const errors=[];page.on('pageerror',e=>errors.push(e.message));await page.goto(server.resolvedUrls.local[0]+'#document=masonry-study');await page.getByRole('button',{name:'Apply construction',exact:true}).waitFor({timeout:120000});await page.waitForFunction(()=>document.querySelector('[aria-label="Construction workspace"]')?.dataset.constructionHash);const old=await page.locator('.construction-workspace').getAttribute('data-construction-hash');await page.getByLabel('Recess (m)',{exact:true}).fill('0.25');await page.getByRole('button',{name:'Apply construction',exact:true}).click();await page.waitForFunction(old=>document.querySelector('.construction-workspace')?.dataset.constructionHash!==old,old);await page.screenshot({path:out+'/desktop.png',fullPage:true});await page.setViewportSize({width:430,height:1000});await page.screenshot({path:out+'/compact.png',fullPage:true});await writeFile(out+'/receipt.json',JSON.stringify({errors,desktop:[1493,1054],compact:[430,1000]},null,2));console.log(JSON.stringify({errors}));}finally{await browser?.close();await server.close();}
