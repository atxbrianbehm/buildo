import {chromium,expect} from "@playwright/test";import {writeFile} from "node:fs/promises";import assert from "node:assert/strict";
const browser=await chromium.launch({channel:"chromium",args:["--enable-unsafe-webgpu"]});
try{const page=await browser.newPage({viewport:{width:1493,height:1054},deviceScaleFactor:1}),errors=[];page.on("pageerror",e=>errors.push(e.message));page.on("console",m=>{if(m.type()==="error")errors.push(m.text());});await page.goto("http://127.0.0.1:5190/#document=masonry-study-ux1",{waitUntil:"domcontentloaded"});await page.waitForFunction(()=>document.querySelector(".construction-workspace")?.dataset.constructionHash&&document.querySelector('[aria-label="Construction inspector"]')?.getAttribute("aria-busy")==="false",{},{timeout:120000});
await expect(page.getByRole("button",{name:"Apply construction",exact:true})).toBeEnabled({timeout:120000});const input=page.getByLabel("Recess (m)",{exact:true});await input.focus();await expect(input).toBeFocused();assert(await input.evaluate(e=>getComputedStyle(e).outlineWidth==="3px"));await page.screenshot({path:"test-results/ux1-p1-2026-10-18/focus-desktop.png",fullPage:true});
await page.getByRole("button",{name:"Apply construction",exact:true}).hover();await page.screenshot({path:"test-results/ux1-p1-2026-10-18/hover-desktop.png",fullPage:true});
const summary=page.getByText("Shape",{exact:true});await summary.focus();await summary.press("Enter");assert(await page.getByLabel("Width (m)",{exact:true}).isVisible());await page.setViewportSize({width:900,height:1100});assert(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth));await page.setViewportSize({width:430,height:1000});assert(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth));assert.deepEqual(errors,[]);
await writeFile("test-results/ux1-p1-2026-10-18/interaction-qa.json",JSON.stringify({schemaVersion:"0.1.0",url:page.url(),focusVisible:true,keyboardAccordion:true,hoverCaptured:true,tabletAndCompactNoOverflow:true,errors},null,2));console.log(JSON.stringify({errors,keyboardAccordion:true}));}finally{await browser.close();}
