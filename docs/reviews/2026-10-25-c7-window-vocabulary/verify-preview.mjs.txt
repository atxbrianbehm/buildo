import {chromium,expect} from "@playwright/test";import {writeFile} from "node:fs/promises";import assert from "node:assert/strict";
const browser=await chromium.launch({channel:"chromium",args:["--enable-unsafe-webgpu"]});
try{const page=await browser.newPage({viewport:{width:1493,height:1054},deviceScaleFactor:1}),errors=[];page.on("pageerror",e=>errors.push({text:e.message,url:"pageerror"}));page.on("console",m=>{if(m.type()==="error")errors.push({text:m.text(),url:m.location().url});});
const response=await page.goto("http://127.0.0.1:5190/#document=c7-window-preview",{waitUntil:"domcontentloaded"});assert.equal(response.status(),200);const apply=page.getByRole("button",{name:"Apply construction",exact:true});await expect(apply).toBeEnabled({timeout:120000});const family=page.getByRole("combobox",{name:"Window assembly",exact:true}),root=page.locator(".construction-workspace"),old=await root.getAttribute("data-construction-hash");await family.focus();await expect(family).toBeFocused();await family.press("ArrowDown");await expect(family).toHaveValue("1");assert.equal(await root.getAttribute("data-construction-hash"),old);assert.equal(await family.evaluate(e=>getComputedStyle(e).outlineWidth),"3px");await page.screenshot({path:"test-results/c7-window-vocabulary-2026-10-25/select-keyboard-focus.png",fullPage:true});await apply.click();await page.waitForFunction(old=>document.querySelector(".construction-workspace")?.dataset.constructionHash!==old,old,{timeout:120000});await expect(apply).toBeEnabled();await page.screenshot({path:"test-results/c7-window-vocabulary-2026-10-25/preview-twin.png",fullPage:true});
assert(errors.every(e=>e.url.endsWith("/favicon.ico")&&e.text.includes("404")));await writeFile("test-results/c7-window-vocabulary-2026-10-25/preview-check.json",JSON.stringify({schemaVersion:"0.1.0",url:page.url(),httpStatus:response.status(),keyboardSelect:true,focusOutlinePx:3,draftBeforeApply:true,actualPublication:true,applicationErrors:[],incidentalFaviconErrors:errors},null,2));console.log(JSON.stringify({httpStatus:200,keyboardSelect:true,actualPublication:true}));}finally{await browser.close();}
