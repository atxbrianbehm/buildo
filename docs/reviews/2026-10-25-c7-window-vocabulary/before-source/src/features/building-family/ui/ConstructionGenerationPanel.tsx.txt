import { DEFAULT_CONSTRUCTION_RECIPE_ID, getConstructionRecipeOptions } from "../core/constructionRecipeCatalog";
import { useRef } from "react";
import type { ConstructionReferenceSnapshot } from "../contracts/constructionGeneration";
import type { ConstructionGenerationControls, ConstructionGenerationLineage } from "../state/constructionGenerationState";

interface Props {
  controls: ConstructionGenerationControls; lineage: ConstructionGenerationLineage | null;
  reference: ConstructionReferenceSnapshot | null; onReferenceFile: (file: File) => void;
  disabled: boolean; onChange: (controls: ConstructionGenerationControls) => void; onGenerate: (reroll: boolean) => void;
}
export function ConstructionGenerationPanel({ controls, lineage, reference, onReferenceFile, disabled, onChange, onGenerate }: Props) {
  const recipes = getConstructionRecipeOptions();
  const recipeId = controls.recipeId ?? lineage?.source.recipe.id ?? DEFAULT_CONSTRUCTION_RECIPE_ID;
  const recipe = recipes.find(item => item.id === recipeId);
  const referenceFile = useRef<HTMLInputElement>(null);
  const text = (field: "buildingSeed" | "materialSeed" | "trimSeed", label: string) => <label><span>{label}</span>
    <input aria-label={label} value={controls[field]} onChange={event => onChange({ ...controls, [field]: event.currentTarget.value })} /></label>;
  return <section className="construction-generation" aria-label="Building generation" data-generation-source={lineage?.source.contentHash}
    data-generation-material-inputs={lineage?.materialPlan.materialInputsHash} data-generation-recipe={lineage?.source.recipe.id}>
    <form onSubmit={event => { event.preventDefault(); onGenerate(false); }}>
      <fieldset disabled={disabled}><legend>Generate a building</legend>
        <p>{recipe?.description} {(reference ?? lineage?.source.reference)?.origin.kind === "utdg-snapshot" ? "Imported UTDG snapshot; coverage and claims are pinned below." : "Local material demo; architectural evidence is unverified."}</p>
        <div className="control-panel__fields">
          <label className="control-panel__field-wide"><span>Construction recipe</span>
            <select aria-label="Construction recipe" value={recipeId} onChange={event => onChange({ ...controls, recipeId: event.currentTarget.value })}>
              {recipes.map(item => <option key={item.id} value={item.id}>{item.label}</option>)}
            </select>
          </label>
        </div>
        <div className="control-panel__fields">
          <label><span>Storeys</span><input type="number" min="2" max="6" step="1" aria-label="Generated storeys" value={controls.floorCount}
            onChange={event => onChange({ ...controls, floorCount: Number(event.currentTarget.value) })} /></label>
          <label><span>Bays per elevation</span><input type="number" min="3" max="6" step="1" aria-label="Generated bays" value={controls.bayCount}
            onChange={event => onChange({ ...controls, bayCount: Number(event.currentTarget.value) })} /></label>
          {text("buildingSeed", "Building seed")}{text("materialSeed", "Material seed")}{text("trimSeed", "Trim seed")}
        </div>
        <div className="control-panel__actions"><button type="button" onClick={() => referenceFile.current?.click()}>Import reference snapshot</button>
          <input ref={referenceFile} type="file" accept=".json,application/json" hidden aria-label="Import reference snapshot file"
            onChange={event => { const file = event.currentTarget.files?.[0]; event.currentTarget.value = ""; if (file) onReferenceFile(file); }} /></div>
        {reference ? <p role="status">Ready for next generation: {reference.id}, revision {reference.revision}. Architecture coverage: {reference.architectureCoverage}. {reference.coverageNotes}</p> : null}
        <div className="construction-generation__locks">
          {([ ["lockFootprint", "Lock footprint"], ["lockStoreys", "Lock storeys"], ["lockOpenings", "Lock openings"], ["preserveAuthored", "Keep edited dimensions"] ] as const).map(([field, label]) =>
            <label key={field}><input type="checkbox" aria-label={label} checked={controls[field]} onChange={event => onChange({ ...controls, [field]: event.currentTarget.checked })} /><span>{label}</span></label>)}
        </div>
        <p>Reroll varies unlocked construction. Edited dimensions and finishes stay yours. A changed material seed replaces the material library and starts a new undo history.</p>
        <div className="control-panel__actions"><button type="submit">Generate building</button>
          <button type="button" onClick={() => onGenerate(true)}>Reroll unlocked</button></div>
      </fieldset>
    </form>
    {lineage ? <details><summary>Generation sources and locks</summary>
      <p>{lineage.source.reference.id} · revision {lineage.source.reference.revision} · {lineage.source.reference.origin.kind}</p>
      <p>Architecture coverage: {lineage.source.reference.architectureCoverage}. {lineage.source.reference.coverageNotes}</p>
      <ul>{lineage.source.reference.claims.map(claim => <li key={claim.id}>{claim.statement} — {claim.basis}, {claim.review}, confidence {claim.confidence ?? "unknown"}; {claim.locator}</li>)}</ul><p>Recipe: {lineage.source.recipe.id} · version {lineage.source.recipe.version}</p>
      <p>{lineage.source.intent.locks.length} retained numeric locks. {lineage.materialPlan.bindings.length} material bindings; roof deck and utility have no reference binding.</p>
      <ul>{lineage.source.intent.locks.map(lock => <li key={lock.semanticPath}>{lock.semanticPath}: {String(lock.lockedValue)} ({lock.reason ?? "Pinned input"})</li>)}</ul>
      <p>Source: <code>{lineage.source.contentHash}</code></p>
      {lineage.source.intentionalExceptions.length ? <ul>{lineage.source.intentionalExceptions.map(item => <li key={item.claimId}>{item.claimId}: {item.reason}</li>)}</ul> : null}
    </details> : null}
  </section>;
}
